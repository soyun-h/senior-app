function listenOnce(fallback, done) {

  let finished = false;

  const fin = (t) => { if (finished) return; finished = true; done(t); };

  const sim = () => setTimeout(() => fin(fallback), 1800);

  if (!SpeechRec) { sim(); return; }

  try {

    const r = new SpeechRec();

    r.lang = "ko-KR"; r.interimResults = false; r.maxAlternatives = 1;

    let heard = false;

    r.onaudiostart = () => { heard = true; };

    r.onresult = (e) => fin(e.results[0][0].transcript);

    r.onerror = (e) => { if (e.error === "no-speech") fin(""); else sim(); };

    r.onend = () => { if (heard) fin(""); else sim(); };

    r.start();

  } catch { sim(); }

/* "오후 3시", "열두 시", "6시 반" → { h, m, pm, raw } */

function readTime(str) {

  const tm = str.match(/(오전|오후|아침|점심|저녁|밤|새벽)?\s*(\d{1,2}|열한|열두|열|한|두|세|네|다섯|여섯|일곱|여덟|아홉)\s*시\s*(반|\d{1,2}\s*분)?/);

  if (!tm) return null;

  const hr = /^\d/.test(tm[2]) ? +tm[2] : HOUR_WORDS[tm[2]];

  const m = tm[3] === "반" ? 30 : tm[3] ? parseInt(tm[3], 10) : 0;

  const ap = tm[1];

  let h, pm;

  if (hr > 12) { pm = true; h = hr - 12; }

  else if (hr === 0) { pm = false; h = 12; }

  else { h = hr; pm = ap && /오전|아침|새벽/.test(ap) ? false : ap ? true : hr === 12 || hr <= 6; }

  return { h, m, pm: !!pm, raw: tm[0] };

}

function findTodoToDelete(raw, todos) {

    .split(/\s+/).map((w) => w.trim()).filter((w) => w.length >= 2);

}

/* 왼쪽으로 밀면 삭제 아이콘이 나타나는 행 (마우스·터치 모두) */

function SwipeRow({ open, onOpen, onClose, onDelete, label, children }) {

  const W = 88;

  const [drag, setDrag] = useState(null);

  const st = useRef({ x: 0, base: 0, down: false, moved: false });

  const x = drag !== null ? drag : open ? -W : 0;

  const down = (e) => { st.current = { x: e.clientX, base: open ? -W : 0, down: true, moved: false }; };

  const move = (e) => {

    const s = st.current; if (!s.down) return;

    const dx = e.clientX - s.x;

    if (!s.moved && Math.abs(dx) > 8) { s.moved = true; try { e.currentTarget.setPointerCapture(e.pointerId); } catch {} }

    if (s.moved) setDrag(Math.max(-W, Math.min(0, s.base + dx)));

  };

  const up = () => {

    const s = st.current; if (!s.down) return;

    s.down = false;

    if (s.moved) { (drag !== null && drag < -W / 2) ? onOpen() : onClose(); setDrag(null); }

  };

  const clickCapture = (e) => {

    if (st.current.moved) { e.stopPropagation(); e.preventDefault(); st.current.moved = false; return; }

    if (open) { e.stopPropagation(); e.preventDefault(); onClose(); }

  };

  return (

    <div className="relative rounded-2xl overflow-hidden" style={{ touchAction: "pan-y" }}>

      <button onClick={onDelete} tabIndex={open ? 0 : -1} aria-label={`${label} 일정 삭제`} className="absolute right-0 top-0 bottom-0 grid place-items-center text-white" style={{ width: W, background: "#D94B3B" }}>

        <Trash2 size={32} />

      </button>

      <div onPointerDown={down} onPointerMove={move} onPointerUp={up} onPointerCancel={up} onClickCapture={clickCapture}

        style={{ transform: `translateX(${x}px)`, transition: drag === null ? "transform .22s ease" : "none" }}>

        {children}

      </div>

    </div>

  );

}

    return () => clearInterval(i);

  }, [alarms, user.name]);


  const applyChar = (id, nk) => {

    const c = charOf(id);

    if (id !== gender || screen === "charSelect") { setLook(c.look); setOwned((o) => [...new Set([...o, ...ownedFor(c)])]); }

    setGender(id); setNick(nk.trim());

  };

  const startService = (id, nk) => {

    applyChar(id, nk); setPoints(1000);

    celebrate("가입 축하 포인트", "1,000p");

  };

  };

  const demo = () => { setUser({ id: "demo", name: "김복순", birth: "1948-03-15", joined: Date.now() }); setTut(1); setScreen("main"); };


  const s = { user, setUser, gender, nick, points, setPoints, look, setLook, owned, setOwned, todos, toggleTodo, addTodo, deleteTodo, journal, addJournal, deleteJournal,

    alarms, setAlarms, editId, setEditId, tut, setTut, go, toast, earn, celebrate, ring, ringing, weather,

    setLoggedOut: () => { setScreen("login"); toast("로그아웃 됐어요"); }, withdraw };


  return (

      <div className="relative w-full max-w-[430px] h-[100dvh] md:h-[860px] md:max-h-[94vh] overflow-hidden md:rounded-[44px] md:border-[10px] md:border-[#33261C] shadow-2xl" style={{ background: CREAM }}>

        {screen === "splash" && <Splash go={go} />}

        {screen === "login" && <Login go={go} demo={demo} />}

        {screen === "signup" && <Signup go={go} setUser={setUser} />}

        {screen === "profile" && <Profile go={go} setUser={setUser} />}

        {screen === "charSelect" && (

          <CharPick title="캐릭터 선택" initialId={gender} initialNick={nick} current={gender} look={look} back={() => go("profile")}
